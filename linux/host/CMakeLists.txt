# nxhost for aarch64 Linux (glibc): the Android host's sources, built as an executable against the stand-ins in linux/shim for the Android NDK interfaces.
# The host's own sources are compiled with __ANDROID__ defined (they select their Linux-kernel paths with it) and NX_LINUX=1 (the few places where Bionic and glibc differ).
#   cmake -S linux/host -B linux/build -G Ninja -DCMAKE_TOOLCHAIN_FILE=linux/host/aarch64-clang.cmake -DGAME=<base game title ID>
cmake_minimum_required(VERSION 3.22)
project(nxhost_linux C CXX ASM)   # PEANUTS(linux): CXX for the companion runtime (dm/peanuts/rt)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64|arm64")
  message(FATAL_ERROR "nxhost runs the game's own ARM64 code: build for aarch64")
endif()
if(NOT GAME)
  message(FATAL_ERROR "GAME is not set: -DGAME=<base game title ID> (the profiles are in dm/games)")
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(HOST ${ROOT}/dm/host)
set(GFX ${ROOT}/dm/gfx)
set(TP ${ROOT}/dm/third_party)
set(SHIM ${ROOT}/linux/shim)
set(AND ${ROOT}/android/host)
set(GEN ${CMAKE_CURRENT_BINARY_DIR}/gen)

set(ASM_IN ${HOST}/entry.S ${GFX}/nvn/nvn_entry.S ${GFX}/nvn/nvn_thunks.S)
set(ASM_OUT)
foreach(s ${ASM_IN})
  get_filename_component(n ${s} NAME_WE)
  set(o ${GEN}/asm/${n}.S)
  add_custom_command(OUTPUT ${o} COMMAND ${Python3_EXECUTABLE} ${AND}/macho_asm_to_elf.py ${s} ${o} DEPENDS ${s} ${AND}/macho_asm_to_elf.py)
  list(APPEND ASM_OUT ${o})
endforeach()

file(GLOB HOST_C CONFIGURE_DEPENDS ${HOST}/*.c)
list(REMOVE_ITEM HOST_C ${HOST}/hle_movie_avc.c)
list(APPEND HOST_C ${HOST}/plat/plat_android.c ${HOST}/plat/movie_avc_android.c ${HOST}/plat/front_android.c)
set(NX_COOP_VENDOR_DIR "${TP}/pokopia-coop" CACHE PATH "The portable co-op module's sources")
set(NX_COOP_VENDOR 0)
if(EXISTS "${NX_COOP_VENDOR_DIR}/coop.h")
  foreach(part ldn vnet mpnet plat auth disc)
    list(APPEND HOST_C "${NX_COOP_VENDOR_DIR}/coop_${part}.c")
  endforeach()
  set(NX_COOP_VENDOR 1)
endif()
file(GLOB HOST_ANDROID_C CONFIGURE_DEPENDS ${HOST}/android/*.c)
list(REMOVE_ITEM HOST_ANDROID_C ${HOST}/android/jni_android.c)   # no Java launcher
list(APPEND HOST_C ${HOST_ANDROID_C})
# PEANUTS begin(linux): the Peanuts host sources, as android/host/CMakeLists.txt adds them (not peanuts_jni.c: no Java launcher). The companion is off on Linux (no second
# screen): its reader and runtime are built and stay idle without a companion package. dm/peanuts/linux/include: <android/log.h> and <jni.h> for these sources only
set(PN ${ROOT}/dm/peanuts)
set(PN_C ${PN}/peanuts.c ${PN}/peanuts_nup.c ${PN}/peanuts_services.c ${PN}/peanuts_graph.c ${PN}/peanuts_frametime.c)
# peanuts_frametime.c without __ANDROID__: its one JNI function (Bridge.frameTimes) is for the Java launcher's graph, which Linux does not have
set_source_files_properties(${PN}/peanuts_frametime.c PROPERTIES COMPILE_OPTIONS -U__ANDROID__)
set(NSP_C ${PN}/peanuts_nsp.c ${PN}/import/nx_nca.c ${PN}/import/nx_crypto.c ${PN}/import/nx_lz4.c)
set_source_files_properties(${NSP_C} PROPERTIES COMPILE_OPTIONS -O2)
list(APPEND PN_C ${NSP_C})
set_source_files_properties(${PN_C} PROPERTIES INCLUDE_DIRECTORIES ${PN}/linux/include)
list(APPEND HOST_C ${PN_C})
add_subdirectory(${PN}/rt ${CMAKE_CURRENT_BINARY_DIR}/peanuts_rt)
# PEANUTS end
file(GLOB GFX_C CONFIGURE_DEPENDS ${GFX}/*.c ${GFX}/nvn/*.c ${GFX}/tex/*.c ${GFX}/vk_android/*.c)
option(NX_LSFG "Lossless Scaling frame generation (GPL-3.0-or-later)" ON)
if(NX_LSFG)
  file(GLOB LSFG_C CONFIGURE_DEPENDS ${GFX}/vk_android/lsfg/*.c)
  list(APPEND GFX_C ${LSFG_C})
endif()
file(GLOB SHIM_C CONFIGURE_DEPENDS ${SHIM}/*.c)

file(GLOB_RECURSE VPX_C CONFIGURE_DEPENDS ${TP}/libvpx/*.c)
add_library(vpx STATIC ${VPX_C})
target_include_directories(vpx PRIVATE ${TP}/libvpx ${TP}/libvpx/config)
target_compile_options(vpx PRIVATE -O2 -w)
add_library(zstddec STATIC ${TP}/zstd/zstddeclib.c)
target_compile_options(zstddec PRIVATE -O2 -w -fvisibility=hidden)
file(GLOB DAV1D_C CONFIGURE_DEPENDS ${TP}/dav1d/src/*.c ${TP}/dav1d/config/tmpl/*.c)
list(FILTER DAV1D_C EXCLUDE REGEX "_tmpl\\.c$")
list(APPEND DAV1D_C ${TP}/dav1d/src/arm/cpu.c)
file(GLOB DAV1D_S CONFIGURE_DEPENDS ${TP}/dav1d/src/arm/64/*.S)
list(FILTER DAV1D_S EXCLUDE REGEX "(_tmpl|/util)\\.S$")
add_library(dav1d STATIC ${DAV1D_C} ${DAV1D_S})
target_include_directories(dav1d PRIVATE ${AND}/dav1d_config ${TP}/dav1d ${TP}/dav1d/include ${TP}/dav1d/config)
target_compile_options(dav1d PRIVATE -O2 -w -fvisibility=hidden)
target_compile_definitions(dav1d PRIVATE _GNU_SOURCE)

# PEANUTS begin(linux): GAME=generic builds Peanuts' generic host (as android/host/CMakeLists.txt's NX_GENERIC_HOST): one nxhost_generic that loads a title's
# profile (DM_PROFILE_PATH, dm/peanuts/profile/gen_runtime_profile.py) and prewarm seed (DM_PROFILE_SEED) at start, no title compiled in
if(GAME STREQUAL "generic")
  set(PROFILE_DIR ${ROOT}/dm/peanuts/profile)
  configure_file(${PROFILE_DIR}/game_profile_generic.h ${GEN}/${GAME}/game_profile.h COPYONLY)
  configure_file(${PROFILE_DIR}/vk_prewarm_seed_generic.h ${GEN}/${GAME}/vk_prewarm_seed.h COPYONLY)
  set(GAME_H ${GEN}/schema_stamp.h)
  add_custom_command(OUTPUT ${GAME_H} COMMAND ${Python3_EXECUTABLE} ${PROFILE_DIR}/gen_runtime_profile.py schema-stamp ${GAME_H}
    DEPENDS ${PROFILE_DIR}/gen_runtime_profile.py ${ROOT}/dm/games/game.py COMMENT "generic host schema stamp")
  set(SEED_H ${PROFILE_DIR}/profile_rt.c)   # (no seed header: the profile loader takes its place in the source list below)
  add_custom_target(nxhost_seed)
else()
  set(GAME_H ${GEN}/${GAME}/game_profile.h)
  add_custom_command(OUTPUT ${GAME_H} COMMAND ${Python3_EXECUTABLE} ${ROOT}/dm/games/game.py header ${GAME} ${GAME_H}
    DEPENDS ${ROOT}/dm/games/${GAME}/game.tsv ${ROOT}/dm/games/game.py COMMENT "game profile ${GAME}")
  set(SEED_H ${GEN}/${GAME}/vk_prewarm_seed.h)
  add_custom_target(nxhost_seed COMMAND ${Python3_EXECUTABLE} ${ROOT}/dm/games/game.py seed ${GAME} ${SEED_H} BYPRODUCTS ${SEED_H} COMMENT "prewarm seed ${GAME}")
endif()
# PEANUTS end
# tls_slot.c first: its variable must be the first of the executable's thread-local block (linux/host/tls_slot.c)
add_executable(nxhost ${CMAKE_CURRENT_SOURCE_DIR}/tls_slot.c ${CMAKE_CURRENT_SOURCE_DIR}/main_linux.c ${CMAKE_CURRENT_SOURCE_DIR}/front_sdl.c ${HOST_C} ${GFX_C} ${SHIM_C} ${ASM_OUT} ${GAME_H} ${SEED_H})
add_dependencies(nxhost nxhost_seed)
target_link_libraries(nxhost PRIVATE peanuts_rt)   # PEANUTS(linux): the companion runtime (C++)
set_target_properties(nxhost PROPERTIES LINKER_LANGUAGE CXX)   # PEANUTS(linux): link with the C++ driver so libstdc++ comes in (the host is C)
set_target_properties(nxhost PROPERTIES OUTPUT_NAME nxhost_${GAME})
target_include_directories(nxhost BEFORE PRIVATE ${SHIM}/include)
# SDL3 (window, gamepads, audio): headers and library of the target system copied into linux/sysroot, else SDL3 built from source (linux/build.sh)
set(NX_SYSROOT ${ROOT}/linux/sysroot CACHE PATH "include/SDL3 and lib/libSDL3.so of the target system")
target_include_directories(nxhost PRIVATE ${NX_SYSROOT}/include)
target_link_directories(nxhost PRIVATE ${NX_SYSROOT}/lib)
# run time: a libSDL3.so.0 in lib/ next to the binary wins, else the system one
set_target_properties(nxhost PROPERTIES BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "\$ORIGIN/lib")
target_include_directories(nxhost PRIVATE ${GEN}/${GAME} ${GEN} ${ROOT}/dm/peanuts/profile ${GFX}/nvn/include ${TP}/vulkan-headers/include ${TP}/libvpx ${TP}/libvpx/config ${TP}/dav1d/include ${TP}/stb_vorbis ${HOST} ${NX_COOP_VENDOR_DIR})   # PEANUTS(linux): ${GEN} and dm/peanuts/profile for the generic host (schema_stamp.h, profile_rt.h)
target_compile_definitions(nxhost PRIVATE _GNU_SOURCE __ANDROID__=1 __ANDROID_API__=33 NX_LINUX=1 main=nx_host_main DM_NUP_ZSTD DM_NSP_DIRECT NX_COOP_VENDOR=${NX_COOP_VENDOR} $<$<BOOL:${NX_LSFG}>:NX_LSFG>)   # PEANUTS(linux): DM_NSP_DIRECT (the reader of dumps is linked: NSP_C above)
target_compile_options(nxhost PRIVATE -O1 -g -Wall -Wno-unused-function -Wno-format -Wno-builtin-macro-redefined -fvisibility=hidden -include ${SHIM}/include/nx_linux_compat.h)
file(GLOB VKA_C ${GFX}/vk_android/*.c)
set_source_files_properties(${VKA_C} PROPERTIES COMPILE_OPTIONS -O2)
# -rdynamic: the host looks some of its own symbols up by name (dlsym); the tables it collects are read through __start_ / __stop_
target_link_options(nxhost PRIVATE -rdynamic -Wl,--allow-shlib-undefined)
target_link_libraries(nxhost PRIVATE vpx dav1d zstddec SDL3 xcb m dl z pthread)   # libvulkan is opened at run time (vka_dev.c)
